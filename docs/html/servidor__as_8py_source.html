50 def handle(self):
51 """
52 ***************************************************************************
53 Função: handle
54
55 @brief Processa uma requisição TCP ao AS.
56
57 Descrição:
58 Lê JSON, valida ação "autenticar", verifica presença de usuário e senha,
59 chama autenticar_no_as e envia resposta ao cliente.
60
61 Parâmetros:
62 Não recebe parâmetros explícitos.
63
64 Valor retornado:
65 @return Não retorna valor.
66
67 Assertiva de entrada:
68 @pre self.request deve conter uma conexão TCP ativa com mensagem JSON válida.
69
70 Assertiva de saída:
71 @post Envia JSON com ok verdadeiro e dados, ou ok falso e erro.
72
73 Exceções:
74 @throws Exception Captura exceções gerais e as converte em resposta de erro.
75
76 Observações:
77 Faz a ponte entre transporte TCP e lógica Kerberos do AS.
78 ***************************************************************************
79 """
80 log_titulo("AS", "Nova conexão recebida no Servidor de Autenticação")
81
82 try:
83 requisicao = receber_json(self.request)
84
85 log_passo("AS", 1, "Requisição recebida via socket")
86 log_dados("AS", "Conteúdo recebido", requisicao)
87
88 acao = requisicao.get("acao")
89
90 if acao != "autenticar":
91 resposta = {
92 "ok": False,
93 "erro": "Ação inválida para o Servidor de Autenticação."
94 }
95 enviar_json(self.request, resposta)
96 log_erro("AS", "Ação inválida recebida.")
97 return
98
99 usuario = requisicao.get("usuario")
100 senha = requisicao.get("senha")
101
102 if not usuario or not senha:
103 resposta = {
104 "ok": False,
105 "erro": "Usuário e senha são obrigatórios."
106 }
107 enviar_json(self.request, resposta)
108 log_erro("AS", "Usuário ou senha não informado.")
109 return
110
111 log_passo(
112 "AS",
113 2,
114 "Validando usuário e senha",
115 "O AS vai buscar o usuário, usar o salt cadastrado e derivar a chave com KDF."
116 )
117
118 log_passo(
119 "AS",
120 3,
121 "Executando autenticação Kerberos no AS",
122 "Se a senha estiver correta, o AS gera a chave Cliente-TGS e o TGT."
123 )
124
125 resposta_as = autenticar_no_as(usuario, senha)
126
127 log_ok("AS", "Senha validada com sucesso.")
128 log_ok("AS", "Chave de sessão Cliente-TGS gerada.")
129 log_ok("AS", "TGT criado e criptografado com a chave secreta do TGS.")
130
131 log_dados("AS", "Resposta criptografada que será enviada ao cliente", resposta_as)
132
133 resposta = {
134 "ok": True,
135 "dados": resposta_as
136 }
137
138 enviar_json(self.request, resposta)
139
140 log_passo(
141 "AS",
142 4,
143 "Resposta enviada ao cliente",
144 "O cliente recebe dados criptografados e um TGT que ele transporta, mas não consegue abrir."
145 )
146
147 except Exception as erro:
148 resposta = {
149 "ok": False,
150 "erro": str(erro)
151 }
152
153 enviar_json(self.request, resposta)
154 log_erro("AS", str(erro))
