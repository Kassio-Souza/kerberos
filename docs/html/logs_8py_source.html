57def _mascarar_texto(valor: str) -> str:
58 """
59 ***************************************************************************
60 Função: _mascarar_texto
61
62 @brief Mascara texto sensível ou longo.
63
64 Descrição:
65 Substitui valores curtos por asteriscos e resume valores longos mantendo
66 apenas prefixo, sufixo e tamanho.
67
68 Parâmetros:
69 @param valor Texto que será mascarado.
70
71 Valor retornado:
72 @return Retorna string mascarada.
73
74 Assertiva de entrada:
75 @pre valor deve ser string.
76
77 Assertiva de saída:
78 @post Retorna texto que não expõe integralmente o valor original.

88 return "***"

93def _eh_campo_sensivel(chave: str) -> bool:
94 """
95 ***************************************************************************
96 Função: _eh_campo_sensivel
97
98 @brief Identifica nomes de campos sensíveis.
99
100 Descrição:
101 Compara a chave em minúsculas com termos associados a senha, chave,
102 verificador, ciphertext, nonce e salt.
103
104 Parâmetros:
105 @param chave Nome do campo a avaliar.
106
107 Valor retornado:
108 @return Retorna True se o campo deve ser mascarado.
109
110 Assertiva de entrada:
111 @pre chave deve ser string.
112
113 Assertiva de saída:
114 @post Retorna booleano usado por _preparar_valor.
115
116 Exceções:
117 @throws Exception Pode propagar AttributeError se chave não for string.
118
119 Observações:
120 A lista é heurística e voltada aos campos usados no projeto.
121 ***************************************************************************
122 """
123 chave = chave.lower()
124
125 campos_sensiveis = [
126 "senha",
127 "password",
128 "chave",
129 "key",
130 "verificador",
131 "ciphertext",
132 "tag",
133 "nonce",
134 "salt",
135 ]
136
137 return any(campo in chave for campo in campos_sensiveis)

140def _preparar_valor(chave, valor):
141 """
142 ***************************************************************************
143 Função: _preparar_valor
144
145 @brief Prepara valores para log seguro.
146
147 Descrição:
148 Percorre recursivamente dicionários e listas, mascarando strings sensíveis
149 ou muito longas.
150
151 Parâmetros:
152 @param chave Nome do campo associado ao valor.
153 @param valor Valor a preparar para exibição.
154
155 Valor retornado:
156 @return Retorna valor sanitizado.
157
158 Assertiva de entrada:
159 @pre valor pode ser dict, list, str ou outro tipo serializável.
160
161 Assertiva de saída:
162 @post Retorna estrutura equivalente com segredos mascarados.
163
164 Exceções:
165 @throws Exception Pode propagar exceções de iteração em estruturas incompatíveis.
166
167 Observações:
168 Preserva formato geral das mensagens Kerberos para fins didáticos.
169 ***************************************************************************
170 """
171 if isinstance(valor, dict):
172 return {k: _preparar_valor(k, v) for k, v in valor.items()}
173
174 if isinstance(valor, list):
175 return [_preparar_valor(chave, item) for item in valor]
176
177 if isinstance(valor, str):
178 if _eh_campo_sensivel(str(chave)):
179 return _mascarar_texto(valor)
180
181 if len(valor) > 80:
182 return _mascarar_texto(valor)
183
184 return valor
185
186 return valor

189def sanitizar(dados):
190 """
191 ***************************************************************************
192 Função: sanitizar
193
194 @brief Remove ou mascara informações sensíveis antes do log.
195
196 Descrição:
197 Aplica _preparar_valor sobre dicionários ou listas e retorna outros tipos sem
198 alteração.
199
200 Parâmetros:
201 @param dados Estrutura a ser preparada para exibição.
202
203 Valor retornado:
204 @return Retorna dados sanitizados.
205
206 Assertiva de entrada:
207 @pre dados pode ser qualquer valor aceito pelos logs.
208
209 Assertiva de saída:
210 @post Estruturas com campos sensíveis têm valores mascarados.
211
212 Exceções:
213 @throws Exception Pode propagar exceções de _preparar_valor.
214
215 Observações:
216 Não altera os dados originais usados pelo protocolo.
217 ***************************************************************************
218 """
219 if isinstance(dados, dict):
220 return {k: _preparar_valor(k, v) for k, v in dados.items()}
221
222 if isinstance(dados, list):
223 return [_preparar_valor("", item) for item in dados]
224
225 return dados

415def log_dados(componente: str, titulo: str, dados):
416 """
417 ***************************************************************************
418 Função: log_dados
419
420 @brief Imprime dados estruturados sanitizados.
421
422 Descrição:
423 Sanitiza dados, serializa em JSON indentado e imprime cada linha com recuo.
424
425 Parâmetros:
426 @param componente Componente responsável pelo log.
427 @param titulo Título do bloco de dados.
428 @param dados Estrutura a exibir.
429
430 Valor retornado:
431 @return Não retorna valor.
432
433 Assertiva de entrada:
434 @pre dados deve ser serializável por json.dumps após sanitização.
435
436 Assertiva de saída:
437 @post Dados são exibidos com campos sensíveis mascarados.
438
439 Exceções:
440 @throws Exception Pode propagar erros de sanitização ou serialização JSON.
441
442 Observações:
443 Útil para apresentação acadêmica dos pacotes sem expor chaves completas.
444 ***************************************************************************
445 """
446 if not MOSTRAR_LOGS:
447 return
448
449 dados_limpos = sanitizar(dados)
450 texto = json.dumps(dados_limpos, indent=4, ensure_ascii=False)
451
452 print(f"[{_hora_atual()}] [{componente}] {titulo}:", flush=True)
453
454 for linha in texto.splitlines():
455 print(f" {linha}", flush=True)
