67def autenticar_no_as(nome_usuario: str, senha: str) -> dict:
68 """
69 ***************************************************************************
70 Função: autenticar_no_as
71
72 @brief Executa a autenticação inicial do cliente no AS.
73
74 Descrição:
75 Busca o usuário cadastrado, deriva a chave a partir da senha e do salt,
76 compara o verificador calculado com o salvo, gera chave Cliente-TGS, cria o
77 TGT e criptografa a resposta destinada ao cliente.
78
79 Parâmetros:
80 @param nome_usuario Nome do usuário que solicita autenticação.
81 @param senha Senha textual informada pelo usuário.
82
83 Valor retornado:
84 @return Retorna dict criptografado com a chave derivada do cliente.
85
86 Assertiva de entrada:
87 @pre nome_usuario != None
88 @pre senha != None
89 @pre O usuário deve existir no cadastro para autenticação bem-sucedida.
90
91 Assertiva de saída:
92 @post Retorna resposta com id_tgs, chave Cliente-TGS e TGT, todos protegidos para
93 @post o cliente; lança exceção se usuário ou senha não forem válidos.
94
95 Exceções:
96 @throws ValueError quando o usuário não existe ou a senha é inválida.
97
98 Observações:
99 O TGT é criptografado com CHAVE_SECRETA_TGS; o cliente o transporta sem
100 conseguir abrir seu conteúdo.
101 ***************************************************************************
102 """
103 dados_usuarios = carregar_usuarios()
104 usuarios = dados_usuarios.get("usuarios", {})
105
106 if nome_usuario not in usuarios:
107 raise ValueError("Usuário não encontrado.")
108
109 dados_usuario = usuarios[nome_usuario]
110
111 salt = dados_usuario["salt"]
112 verificador_salvo = dados_usuario["verificador"]
113
114 chave_cliente = derivar_chave_senha(senha, salt)
115 verificador_calculado = gerar_verificador_chave(chave_cliente)
116
117 if verificador_calculado != verificador_salvo:
118 raise ValueError("Senha inválida.")
119
120 chave_sessao_cliente_tgs = gerar_chave_simetrica()
121 chave_sessao_cliente_tgs_base64 = bytes_para_base64(chave_sessao_cliente_tgs)
122
123 tgt = criar_tgt(
124 id_cliente=nome_usuario,
125 chave_sessao_cliente_tgs_base64=chave_sessao_cliente_tgs_base64,
126 id_tgs="tgs"
127 )
128
129 tgt_criptografado = criptografar_json(
130 CHAVE_SECRETA_TGS,
131 tgt
132 )
133
134 resposta_para_cliente = {
135 "id_tgs": "tgs",
136 "chave_sessao_cliente_tgs": chave_sessao_cliente_tgs_base64,
137 "tgt": tgt_criptografado
138 }
139
140 resposta_criptografada = criptografar_json(
141 chave_cliente,
142 resposta_para_cliente
143 )
144
145 return resposta_criptografada
