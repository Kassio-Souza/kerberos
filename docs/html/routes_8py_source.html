138def autenticar_com_kerberos(usuario: str, senha: str) -> dict:
139 """
140 ***************************************************************************
141 Função: autenticar_com_kerberos
142
143 @brief Executa o fluxo Kerberos inicial via sockets.
144
145 Descrição:
146 Chama o AS, deriva a chave do cliente com senha e salt, abre a resposta do
147 AS, cria autenticador para o TGS, solicita ticket de serviço e abre a parte
148 da resposta destinada ao cliente.
149
150 Parâmetros:
151 @param usuario Nome do usuário do login.
152 @param senha Senha informada no formulário.
153
154 Valor retornado:
155 @return Retorna dict com ticket_servico e chave_sessao_servico.
156
157 Assertiva de entrada:
158 @pre usuario != None
159 @pre senha != None
160 @pre AS e TGS devem estar disponíveis nas portas configuradas.
161
162 Assertiva de saída:
163 @post Retorna credenciais para acessar o serviço de notas.
164
165 Exceções:
166 @throws Exception Propaga ValueError de autenticação, erros de rede e erros de descriptografia.
167
168 Observações:
169 Esta função concentra as etapas cliente do AS-REQ/AS-REP e TGS-REQ/TGS-REP.
170 ***************************************************************************
171 """
172
173 log_titulo("CLIENTE WEB", "Iniciando fluxo Kerberos completo")
174
175 log_passo(
176 "CLIENTE WEB",
177 1,
178 "Usuário informou login e senha",
179 "A senha será usada apenas para derivar a chave do cliente."
180 )
181
182 resposta_as_criptografada = chamar_as_autenticacao(usuario, senha)
183
184 log_passo(
185 "CLIENTE WEB",
186 2,
187 "Buscando salt do usuário",
188 "O salt não é secreto. Ele serve para derivar a chave a partir da senha."
189 )
190
191 salt = obter_salt_usuario(usuario)
192
193 log_dados(
194 "CLIENTE WEB",
195 "Salt encontrado para derivação da chave",
196 {"usuario": usuario, "salt": salt}
197 )
198
199 log_passo(
200 "CLIENTE WEB",
201 3,
202 "Derivando chave do cliente com KDF",
203 "Aqui é usada a senha + salt para gerar a chave simétrica do cliente."
204 )
205
206 chave_cliente = derivar_chave_senha(senha, salt)
207
208 log_ok("CLIENTE WEB", "Chave do cliente derivada com sucesso usando KDF.")
209
210 log_passo(
211 "CLIENTE WEB",
212 4,
213 "Abrindo resposta do AS",
214 "O cliente consegue abrir porque conhece a senha correta."
215 )
216
217 resposta_as = descriptografar_json(chave_cliente, resposta_as_criptografada)
218
219 log_dados("CLIENTE WEB", "Resposta aberta do AS", resposta_as)
220
221 chave_sessao_cliente_tgs = resposta_as["chave_sessao_cliente_tgs"]
222 tgt = resposta_as["tgt"]
223
224 log_ok("CLIENTE WEB", "Chave de sessão Cliente-TGS obtida.")
225 log_ok("CLIENTE WEB", "TGT recebido. O cliente transporta o TGT, mas não consegue abrir seu conteúdo.")
226
227 log_passo(
228 "CLIENTE WEB",
229 5,
230 "Criando autenticador para o TGS",
231 "Esse autenticador prova que o cliente conhece a chave Cliente-TGS."
232 )
233
234 autenticador_tgs = criar_autenticador(
235 usuario,
236 chave_sessao_cliente_tgs
237 )
238
239 log_dados("CLIENTE WEB", "Autenticador Cliente-TGS criado", autenticador_tgs)
240
241 resposta_tgs = chamar_tgs_emitir_ticket(
242 usuario,
243 tgt,
244 "notas",
245 autenticador_tgs
246 )
247
248 log_passo(
249 "CLIENTE WEB",
250 6,
251 "Abrindo resposta do TGS",
252 "O cliente abre a parte criptografada com a chave Cliente-TGS."
253 )
254
255 dados_cliente = descriptografar_json(
256 base64_para_bytes(chave_sessao_cliente_tgs),
257 resposta_tgs["resposta_cliente"]
258 )
259
260 log_dados("CLIENTE WEB", "Dados abertos da resposta do TGS", dados_cliente)
261
262 log_ok("CLIENTE WEB", "Chave de sessão Cliente-Serviço obtida.")
263 log_ok("CLIENTE WEB", "Ticket do Serviço de Notas recebido.")
264
265 log_passo(
266 "CLIENTE WEB",
267 7,
268 "Fluxo inicial concluído",
269 "Agora o cliente pode acessar o Serviço de Notas usando ticket + autenticador."
270 )
271
272 return {
273 "ticket_servico": resposta_tgs["ticket_servico"],
274 "chave_sessao_servico": dados_cliente["chave_sessao_cliente_servico"],
275 }

279def validar_autenticacao_mutua(
280 chave_sessao_servico: str,
281 ap_rep: dict,
282 timestamp_enviado: int,
283 nonce_enviado: str,
284) -> bool:
285 """
286 ***************************************************************************
287 Função: validar_autenticacao_mutua
288
289 @brief Valida o AP-REP retornado pelo serviço de notas.
290
291 Descrição:
292 Descriptografa o AP-REP com a chave Cliente-Serviço e confirma se o serviço
293 devolveu timestamp incrementado e o mesmo nonce enviado pelo cliente.
294
295 Parâmetros:
296 @param chave_sessao_servico Chave Cliente-Serviço em Base64.
297 @param ap_rep Resposta criptografada de autenticação mútua.
298 @param timestamp_enviado Timestamp usado no autenticador do cliente.
299 @param nonce_enviado Nonce usado no autenticador do cliente.
300
301 Valor retornado:
302 @return Retorna True quando a autenticação mútua é confirmada.
303
304 Assertiva de entrada:
305 @pre ap_rep deve estar cifrado com a chave Cliente-Serviço correta.
306
307 Assertiva de saída:
308 @post Retorna True ou lança ValueError em inconsistência.
309
310 Exceções:
311 @throws ValueError se AP-REP, timestamp ou nonce não forem confirmados.
312
313 Observações:
314 Confirma que o serviço conhece a chave de sessão e, portanto, é o serviço esperado.
315 ***************************************************************************
316 """
317
318 log_passo(
319 "CLIENTE WEB",
320 8,
321 "Validando autenticação mútua",
322 "O cliente vai abrir o AP-REP para confirmar se o serviço conhece a chave Cliente-Serviço."
323 )
324
325 chave_bytes = base64_para_bytes(chave_sessao_servico)
326 ap_rep_aberto = descriptografar_json(chave_bytes, ap_rep)
327
328 log_dados("CLIENTE WEB", "AP-REP aberto pelo cliente", ap_rep_aberto)
329
330 if not ap_rep_aberto:
331 raise ValueError("Falha na autenticação mútua: resposta AP-REP inválida.")
332
333 if ap_rep_aberto.get("timestamp_confirmado") != timestamp_enviado + 1:
334 raise ValueError("Falha na autenticação mútua: timestamp não confirmado.")
335
336 if ap_rep_aberto.get("nonce_confirmado") != nonce_enviado:
337 raise ValueError("Falha na autenticação mútua: nonce não confirmado.")
338
339 log_ok("CLIENTE WEB", "Autenticação mútua confirmada.")
340 log_ok("CLIENTE WEB", "O serviço provou que conhece a chave de sessão Cliente-Serviço.")
341
342 return True

345def listar_notas_protegidas(
346 usuario: str,
347 ticket_servico: dict,
348 chave_sessao_servico: str,
349) -> list:
350 """
351 ***************************************************************************
352 Função: listar_notas_protegidas
353
354 @brief Lista notas usando ticket e autenticador Kerberos.
355
356 Descrição:
357 Cria autenticador Cliente-Serviço com timestamp e nonce, chama o Serviço de
358 Notas e valida o AP-REP antes de retornar a lista.
359
360 Parâmetros:
361 @param usuario Usuário autenticado na sessão web.
362 @param ticket_servico Ticket emitido pelo TGS.
363 @param chave_sessao_servico Chave Cliente-Serviço em Base64.
364
365 Valor retornado:
366 @return Retorna lista de notas.
367
368 Assertiva de entrada:
369 @pre ticket_servico e chave_sessao_servico devem estar na sessão Flask.
370
371 Assertiva de saída:
372 @post Retorna notas apenas após autenticação mútua válida.
373
374 Exceções:
375 @throws Exception Propaga erros do serviço, rede, criptografia ou validação AP-REP.
376
377 Observações:
378 A função implementa a etapa AP-REQ/AP-REP do cliente para listagem.
379 ***************************************************************************
380 """
381
382 timestamp_enviado = timestamp_atual()
383 nonce_enviado = uuid.uuid4().hex
384
385 autenticador_servico = criar_autenticador(
386 usuario,
387 chave_sessao_servico,
388 nonce=nonce_enviado,
389 timestamp=timestamp_enviado,
390 )
391
392 resultado = chamar_servico_notas_listar(
393 usuario,
394 ticket_servico,
395 autenticador_servico
396 )
397
398 validar_autenticacao_mutua(
399 chave_sessao_servico,
400 resultado["ap_rep"],
401 timestamp_enviado,
402 nonce_enviado,
403 )
404
405 return resultado.get("notas", [])

608 @app.route("/notas", methods=["GET", "POST"])
609 def notas():
610 """
611 ***************************************************************************
612 Função: notas
613
614 @brief Exibe notas e processa criação de notas.
615
616 Descrição:
617 Valida sessão Kerberos, lista notas em GET e, em POST, envia uma ou
618 mais notas ao serviço protegido quando o usuário é professor.
619
620 Parâmetros:
621 Não recebe parâmetros explícitos.
622
623 Valor retornado:
624 @return Retorna HTML renderizado ou redirecionamento Flask.
625
626 Assertiva de entrada:
627 @pre Sessão deve conter usuario, ticket_servico e chave_sessao_servico.
628
629 Assertiva de saída:
630 @post Exibe notas autenticadas ou registra notas via serviço protegido.
631
632 Exceções:
633 @throws Exception Captura exceções de acesso ao serviço e renderiza página de erro.
634
635 Observações:
636 A rota usa AP-REQ/AP-REP a cada operação protegida.
637 ***************************************************************************
638 """
639 if "usuario" not in session:
640 flash("Faça login para acessar o sistema de notas.")
641 return redirect(url_for("login"))
642
643 usuario = session["usuario"]
644 tipo_usuario = obter_tipo_usuario(usuario)
645 ticket_servico = session.get("ticket_servico")
646 chave_sessao_servico = session.get("chave_sessao_servico")
647
648 if not ticket_servico or not chave_sessao_servico:
649 flash("Sessão Kerberos inválida. Faça login novamente.")
650 return redirect(url_for("logout"))
651
652 try:
653 if request.method == "POST":
654 if not usuario_e_professor(usuario):
655 flash("Erro: apenas professores podem lançar ou alterar notas escolares.")
656 return redirect(url_for("notas"))
657
658 aluno = request.form.get("aluno", "").strip()
659
660 disciplinas = request.form.getlist("disciplina[]")
661 disciplinas_extras = request.form.getlist("disciplina_extra[]")
662 valores = request.form.getlist("valor[]")
663
664 if not aluno:
665 flash("Selecione o aluno.")
666 return redirect(url_for("notas"))
667
668 quantidade_lancada = 0
669
670 for indice, disciplina_opcao in enumerate(disciplinas):
671 disciplina_opcao = disciplina_opcao.strip()
672
673 disciplina_extra = ""
674 valor = ""
675
676 if indice < len(disciplinas_extras):
677 disciplina_extra = disciplinas_extras[indice].strip()
678
679 if indice < len(valores):
680 valor = valores[indice].strip()
681
682 if disciplina_opcao == "__outra__":
683 disciplina = disciplina_extra
684 else:
685 disciplina = disciplina_opcao
686
687 if not disciplina or not valor:
688 continue
689
690 criar_nota_protegida(
691 usuario,
692 aluno,
693 disciplina,
694 valor,
695 ticket_servico,
696 chave_sessao_servico
697 )
698
699 quantidade_lancada += 1
700
701 if quantidade_lancada == 0:
702 flash("Informe pelo menos uma disciplina e uma nota.")
703 return redirect(url_for("notas"))
704
705 flash(f"{quantidade_lancada} nota(s) lançada(s) com sucesso.")
706 return redirect(url_for("notas"))
707
708 lista_notas = listar_notas_protegidas(
709 usuario,
710 ticket_servico,
711 chave_sessao_servico
712 )
713
714 return render_template(
715 "notas.html",
716 usuario=usuario,
717 tipo_usuario=tipo_usuario,
