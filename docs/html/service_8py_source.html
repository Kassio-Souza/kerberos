35def _autenticar_no_servico(
36 usuario: str,
37 ticket_servico_criptografado: dict,
38 autenticador_criptografado: dict,
39) -> dict:
40 """
41 ***************************************************************************
42 Função: _autenticar_no_servico
43
44 @brief Valida ticket de serviço e autenticador Cliente-Serviço.
45
46 Descrição:
47 Abre o ticket emitido pelo TGS, confirma o usuário, descriptografa o
48 autenticador com a chave Cliente-Serviço, verifica timestamp e nonce contra
49 replay e gera AP-REP para autenticação mútua.
50
51 Parâmetros:
52 @param usuario Usuário declarado na operação protegida.
53 @param ticket_servico_criptografado Ticket de serviço recebido do cliente.
54 @param autenticador_criptografado Autenticador Cliente-Serviço.
55
56 Valor retornado:
57 @return Retorna dict AP-REP criptografado com a chave Cliente-Serviço.
58
59 Assertiva de entrada:
60 @pre usuario != None
61 @pre ticket_servico_criptografado deve ter sido emitido para o serviço notas.
62 @pre autenticador_criptografado deve conter usuário, timestamp e nonce válidos.
63
64 Assertiva de saída:
65 @post Retorna AP-REP com timestamp_confirmado e nonce_confirmado quando a validação
66 @post é bem-sucedida.
67
68 Exceções:
69 @throws ValueError para ticket de outro usuário, autenticador ausente,
70
71 Observações:
72 NONCES_USADOS guarda chaves de replay em memória durante a execução do serviço.
73 ***************************************************************************
74 """
75 ticket_servico = abrir_ticket_servico(NOME_SERVICO, ticket_servico_criptografado)
76
77 if ticket_servico.get("usuario") != usuario:
78 raise ValueError("Ticket pertence a outro usuário.")
79
80 chave_sessao_base64 = ticket_servico["chave_sessao_cliente_servico"]
81
82 if not autenticador_criptografado:
83 raise ValueError("Autenticador não informado.")
84
85 try:
86 autenticador = abrir_autenticador(chave_sessao_base64, autenticador_criptografado)
87 except Exception as erro:
88 raise ValueError(
89 "Autenticador inválido ou não pode ser descriptografado."
90 ) from erro
91
92 if autenticador.get("usuario") != usuario:
93 raise ValueError("Usuário do autenticador diferente do usuário do ticket.")
94
95 timestamp = autenticador.get("timestamp")
96 if timestamp is None:
97 raise ValueError("Autenticador sem timestamp.")
98
99 if ticket_expirou(timestamp, TEMPO_MAXIMO_AUTENTICADOR):
100 raise ValueError("Autenticador expirado.")
101
102 if timestamp > timestamp_atual() + TEMPO_MAXIMO_AUTENTICADOR:
103 raise ValueError("Autenticador com timestamp inválido.")
104
105 nonce = autenticador.get("nonce")
106
107 if not nonce:
108 raise ValueError("Autenticador sem nonce.")
109
110 chave_replay = f"{usuario}:{timestamp}:{nonce}"
111
112 if chave_replay in NONCES_USADOS:
113 raise ValueError("Autenticador rejeitado: possível tentativa de replay.")
114
115 NONCES_USADOS.add(chave_replay)
116
117 chave_sessao_bytes = base64_para_bytes(chave_sessao_base64)
118
119 ap_rep = criptografar_json(
120 chave_sessao_bytes,
121 {
122 "timestamp_confirmado": timestamp + 1,
123 "nonce_confirmado": nonce,
124 }
125 )
126
127 return ap_rep

130def listar_notas(
131 usuario: str,
132 ticket_servico_criptografado: dict,
133 autenticador_criptografado: dict,
134) -> dict:
135 """
136 ***************************************************************************
137 Função: listar_notas
138
139 @brief Lista notas após autenticação Kerberos no serviço.
140
141 Descrição:
142 Autentica a requisição, decide se o usuário é professor ou aluno e consulta
143 todas as notas ou apenas as notas do próprio usuário.
144
145 Parâmetros:
146 @param usuario Usuário autenticado que solicita a listagem.
147 @param ticket_servico_criptografado Ticket de serviço emitido pelo TGS.
148 @param autenticador_criptografado Autenticador Cliente-Serviço.
149
150 Valor retornado:
151 @return Retorna dict com lista de notas e AP-REP.
152
153 Assertiva de entrada:
154 @pre Credenciais Kerberos devem ser válidas para o serviço notas.
155
156 Assertiva de saída:
157 @post Professor recebe todas as notas; aluno recebe somente suas notas.
158
159 Exceções:
160 @throws Exception Propaga ValueError de _autenticar_no_servico e erros do repositório.
161
162 Observações:
163 Combina autenticação Kerberos com autorização simples baseada no tipo de usuário.
164 ***************************************************************************
165 """
166 ap_rep = _autenticar_no_servico(
167 usuario,
168 ticket_servico_criptografado,
169 autenticador_criptografado
170 )
171
172 if usuario_e_professor(usuario):
173 notas = listar_todas_notas()
174 else:
175 notas = listar_notas_usuario(usuario)
176
177 return {
178 "notas": notas,
179 "ap_rep": ap_rep
180 }

195 @brief Cria uma nota escolar após autenticação e autorização.
196
197 Descrição:
198 Valida credenciais Kerberos, exige que o usuário seja professor, normaliza os

206 @param ticket_servico_criptografado Ticket emitido pelo TGS.
207 @param autenticador_criptografado Autenticador Cliente-Serviço.
208
209 Valor retornado:
210 @return Retorna dict com a nota criada e AP-REP.
211
212 Assertiva de entrada:
213 @pre usuario deve ser professor.
214 @pre aluno, disciplina e valor devem conter texto após strip.

216
217 Assertiva de saída:
218 @post Persiste a nota e retorna os dados cadastrados.
219
220 Exceções:
221 @throws ValueError se usuário não for professor ou campos obrigatórios faltarem.
222
223 Observações:
224 A função separa autenticação Kerberos de autorização acadêmica por papel.
225 ***************************************************************************
226 """
227 ap_rep = _autenticar_no_servico(
228 usuario,
229 ticket_servico_criptografado,
230 autenticador_criptografado
231 )
232
233 if not usuario_e_professor(usuario):
234 raise ValueError("Apenas professores podem cadastrar ou alterar notas.")
235 aluno = aluno.strip()
236 disciplina = disciplina.strip()
237 valor = valor.strip()
238
239 if not aluno or not disciplina or not valor:

241
242 nova_nota = adicionar_nota_aluno(
243 aluno=aluno,
244 disciplina=disciplina,
245 valor=valor,
246 professor=usuario,

249 return {
